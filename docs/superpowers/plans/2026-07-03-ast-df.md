# AST-DF Webshell Detection Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an optional AST-DF-style detection layer and offline training path to the existing webshell scanner without changing current scan behavior when no model is present.

**Architecture:** Add one focused module, `libs/ast_df.py`, that extracts stdlib-only AST-like static features, loads a trusted local pickle model only when configured and present, and returns `(subscore, reason)`. Wire it into the scanner as evidence group `ast_df` after existing static heuristics, with policy caps matching the existing score-group pattern. Add an optional training script that consumes local or explicitly downloaded datasets, trains Deep Forest when available or sklearn fallback when installed, and writes `models/ast_df.pkl`.

**Tech Stack:** Python 3 stdlib (`argparse`, `hashlib`, `os`, `pickle`, `re`, `shutil`, `sys`, `urllib.request`, `zipfile`); optional training-only imports (`deepforest`/`deepforest.CascadeForestClassifier` if installed, otherwise `sklearn.ensemble.RandomForestClassifier`). No new mandatory runtime package.

## Global Constraints

- Add an AST-DF-style detection layer without breaking the current YARA/IOC/SSDeep/heuristic flow.
- Scanner must remain usable when no AST-DF model exists.
- Dataset download and model training must be explicit tooling, not something the scanner does during a scan.
- Use `libs/ast_df.py` for static feature extraction, local model loading, and scanner scoring.
- Run AST-DF after existing static heuristics.
- Add evidence group `ast_df` when local model predicts webshell probability above policy threshold.
- Cap AST-DF score with `ast_df_score_cap`.
- Add policy defaults in both `webshell-scan-docker/config/scan-policy.json` and `webshell-scan-docker/libs/scan_policy.py`:
  - `ast_df_enabled`: true
  - `ast_df_model_path`: `models/ast_df.pkl`
  - `ast_df_min_probability`: `0.70`
  - `ast_df_score_cap`: `90`
  - `ast_df_max_file_bytes`: `2097152`
- Do not execute samples.
- Do not auto-download during scan.
- Downloads are explicit in the train tool only.
- Unknown external datasets are not committed to the repo.
- Model loading is local-path only.
- Diff must be small; do not refactor scanner flow outside AST-DF integration.
- Keep config defaults aligned; `tests/check_scan_policy_defaults.py` must pass.

```python
# -*- coding: utf-8 -*-
"""Optional AST-DF-style static features and local model scoring."""

import os
import pickle
import re

_SUPPORTED_EXT = {
    ".php", ".phtml", ".php5", ".php7", ".phar", ".inc",
    ".jsp", ".jspx", ".asp", ".aspx", ".ashx", ".asmx",
}

_TOKEN_RE = re.compile(
    r"\$[A-Za-z_][A-Za-z0-9_]*|[A-Za-z_][A-Za-z0-9_]*|[{}()\[\];,.]|['\"]",
    re.DOTALL,
)

_PATTERNS = {
    "source_count": [
        r"\$_(?:GET|POST|REQUEST|COOKIE|SERVER|FILES)\s*\[",
        r"php://input",
        r"filter_input\s*\(",
        r"getallheaders\s*\(",
        r"request\s*\.\s*(?:getParameter|getParameterMap|getHeader|getInputStream|getReader)\s*\(",
        r"Request\s*\.\s*(?:QueryString|Form|Params|Cookies|ServerVariables|BinaryRead|InputStream)",
        r"Request\s*\[",
    ],
    "sink_count": [
        r"\b(?:eval|assert|system|shell_exec|passthru|proc_open|popen|exec)\s*\(",
        r"Runtime\s*\.\s*getRuntime\s*\(\s*\)\s*\.\s*exec",
        r"new\s+ProcessBuilder",
        r"Process\s*\.\s*Start",
        r"ProcessStartInfo",
        r"\b(?:Eval|Execute|Shell)\s*\(",
        r"cmd\.exe|powershell",
    ],
    "decode_count": [
        r"base64_decode\s*\(",
        r"gzinflate\s*\(",
        r"gzuncompress\s*\(",
        r"str_rot13\s*\(",
        r"urldecode\s*\(",
        r"FromBase64String\s*\(",
        r"java\.util\.Base64",
        r"URLDecoder\s*\.\s*decode\s*\(",
    ],
    "dynamic_dispatch_count": [
        r"\$[A-Za-z_][A-Za-z0-9_]*\s*\(",
        r"call_user_func(?:_array)?\s*\(",
        r"create_function\s*\(",
        r"Class\s*\.\s*forName",
        r"getDeclaredMethod\s*\(",
        r"\.\s*invoke\s*\(",
        r"Assembly\s*\.\s*Load",
        r"InvokeMember\s*\(",
    ],
    "file_ops_count": [
        r"file_get_contents\s*\(",
        r"file_put_contents\s*\(",
        r"fopen\s*\(",
        r"move_uploaded_file\s*\(",
        r"delete\s*\(",
        r"File(?:Input|Output)Stream",
        r"Scripting\.FileSystemObject",
    ],
    "network_count": [
        r"curl_exec\s*\(",
        r"fsockopen\s*\(",
        r"stream_socket_client\s*\(",
        r"Socket\s*\(",
        r"WebClient",
        r"HttpWebRequest",
    ],
    "auth_channel_count": [
        r"password",
        r"passwd",
        r"pwd",
        r"token",
        r"auth",
        r"md5\s*\(",
        r"sha1\s*\(",
    ],
}

FEATURE_NAMES = (
    "file_size",
    "line_count",
    "token_count",
    "max_depth",
    "branch_count",
    "source_count",
    "sink_count",
    "decode_count",
    "dynamic_dispatch_count",
    "file_ops_count",
    "network_count",
    "auth_channel_count",
    "long_string_count",
    "string_concat_count",
    "variable_variable_count",
    "non_alnum_ratio",
    "suspicious_score",
    "is_php",
    "is_jsp",
    "is_aspnet",
)


def _decode_text(data):
    if data is None:
        return ""
    if isinstance(data, str):
        return data
    for encoding in ("utf-8", "latin-1"):
        try:
            return data.decode(encoding, errors="replace")
        except Exception:
            pass
    return ""


def _count(patterns, text):
    total = 0
    for pattern in patterns:
        total += len(re.findall(pattern, text, re.IGNORECASE | re.DOTALL))
    return total


def _max_depth(text):
    depth = 0
    max_seen = 0
    for char in text:
        if char in "({[":
            depth += 1
            if depth > max_seen:
                max_seen = depth
        elif char in ")}]" and depth:
            depth -= 1
    return max_seen


def _model_path(policy, app_path):
    configured = str((policy or {}).get("ast_df_model_path", "models/ast_df.pkl"))
    if os.path.isabs(configured):
        return configured
    base = app_path or os.getcwd()
    return os.path.join(base, configured)


def _predict_probability(model, vector):
    if hasattr(model, "predict_proba"):
        probabilities = model.predict_proba([vector])
        first = probabilities[0]
        if len(first) > 1:
            return float(first[1])
        return float(first[0])
    if hasattr(model, "predict"):
        prediction = model.predict([vector])[0]
        return float(prediction)
    return 0.0


def extract_features(file_data, extension, file_path="", policy=None):
    """Extract stdlib-only AST-DF-style numeric features from source bytes."""
    text = _decode_text(file_data)
    ext = (extension or os.path.splitext(file_path or "")[1]).lower()
    tokens = _TOKEN_RE.findall(text)
    size = len(file_data or b"") if not isinstance(file_data, str) else len(file_data.encode("utf-8", "ignore"))

    features = {
        "file_size": float(size),
        "line_count": float(max(1, text.count("\n") + 1)),
        "token_count": float(len(tokens)),
        "max_depth": float(_max_depth(text)),
        "branch_count": float(_count([r"\b(?:if|else|for|foreach|while|switch|case|try|catch)\b"], text)),
        "long_string_count": float(_count([r"['\"][A-Za-z0-9+/=]{80,}['\"]"], text)),
        "string_concat_count": float(_count([r"['\"]\s*[.+&]\s*['\"]"], text)),
        "variable_variable_count": float(_count([r"\$\$[A-Za-z_][A-Za-z0-9_]*", r"\$\{\s*\$"], text)),
        "non_alnum_ratio": 0.0,
        "is_php": 1.0 if ext in {".php", ".phtml", ".php5", ".php7", ".phar", ".inc"} else 0.0,
        "is_jsp": 1.0 if ext in {".jsp", ".jspx"} else 0.0,
        "is_aspnet": 1.0 if ext in {".asp", ".aspx", ".ashx", ".asmx"} else 0.0,
    }

    if text:
        non_alnum = sum(1 for char in text if not char.isalnum() and not char.isspace())
        features["non_alnum_ratio"] = float(non_alnum) / float(len(text))

    for name, patterns in _PATTERNS.items():
        features[name] = float(_count(patterns, text))

    features["suspicious_score"] = float(
        features["source_count"] * 2
        + features["sink_count"] * 3
        + features["decode_count"] * 2
        + features["dynamic_dispatch_count"] * 2
        + features["file_ops_count"]
        + features["network_count"]
        + features["auth_channel_count"]
        + features["long_string_count"]
        + features["variable_variable_count"] * 2
    )
    return features


def feature_vector(features):
    """Return model input values in stable training/inference order."""
    return [float(features.get(name, 0.0)) for name in FEATURE_NAMES]


def ast_df_heuristic(file_data, extension, file_path, policy, app_path=None):
    """Score a file with a trusted local AST-DF model when one is configured."""
    policy = policy or {}
    if not policy.get("ast_df_enabled", True):
        return 0, None

    ext = (extension or os.path.splitext(file_path or "")[1]).lower()
    if ext not in _SUPPORTED_EXT:
        return 0, None

    max_bytes = int(policy.get("ast_df_max_file_bytes", 2 * 1024 * 1024))
    size = len(file_data or b"") if not isinstance(file_data, str) else len(file_data.encode("utf-8", "ignore"))
    if size > max_bytes:
        return 0, None

    path = _model_path(policy, app_path)
    if not os.path.isfile(path):
        return 0, None

    try:
        with open(path, "rb") as handle:
            model = pickle.load(handle)
        features = extract_features(file_data, ext, file_path, policy)
        probability = _predict_probability(model, feature_vector(features))
    except Exception:
        return 0, None

    threshold = float(policy.get("ast_df_min_probability", 0.70))
    if probability < threshold:
        return 0, None

    cap = int(policy.get("ast_df_score_cap", 90))
    score = int(round(min(cap, max(1, probability * cap))))
    reason = "AST-DF model probability %.2f (static AST-like features)" % probability
    return score, reason
```

```python
# -*- coding: utf-8 -*-
"""Train optional AST-DF model from local or explicitly downloaded samples."""

import argparse
import os
import pickle
import shutil

import urllib.parse
import urllib.request

    ".jsp", ".jspx", ".asp", ".aspx", ".ashx", ".asmx",
}


def parse_args():
    parser = argparse.ArgumentParser(
        description="Train optional AST-DF webshell model from static source samples.")
    parser.add_argument(
        "--dataset-dir", default=os.path.join(APP_ROOT, "dataset", "ast-df"),
        help="Dataset root containing webshell/ and benign/ directories.")
    parser.add_argument(
        "--output", default=os.path.join(APP_ROOT, "models", "ast_df.pkl"),

    parser.add_argument(
        "--download-url", action="append", default=[],
        help="Explicit .zip dataset URL to download before training. Host must be allowlisted.")
    parser.add_argument(
        "--allow-host", action="append", default=[],
        help="Allowed download host. Required for each --download-url host.")
    return parser.parse_args()


def download_archives(urls, allowed_hosts, dataset_dir):
    if not urls:
        return
    allowed = set(allowed_hosts or [])
    download_dir = os.path.join(dataset_dir, "downloads")
    os.makedirs(download_dir, exist_ok=True)
    for index, url in enumerate(urls, 1):
        parsed = urllib.parse.urlparse(url)
        if parsed.scheme not in ("https", "http") or parsed.hostname not in allowed:
            raise SystemExit("download host not allowlisted: %s" % url)
        if not parsed.path.lower().endswith(".zip"):
            raise SystemExit("only .zip dataset downloads are supported: %s" % url)
        archive_path = os.path.join(download_dir, "dataset-%d.zip" % index)
        print("Downloading %s" % url)
        urllib.request.urlretrieve(url, archive_path)
        extract_dir = os.path.join(download_dir, "dataset-%d" % index)
        if os.path.isdir(extract_dir):
            shutil.rmtree(extract_dir)
        os.makedirs(extract_dir, exist_ok=True)
        with zipfile.ZipFile(archive_path) as archive:
            archive.extractall(extract_dir)
        print("Extracted %s" % extract_dir)


def iter_samples(root):
    for label_name, label in (("benign", 0), ("webshell", 1)):
        label_dir = os.path.join(root, label_name)
        if not os.path.isdir(label_dir):
            raise SystemExit("missing dataset directory: %s" % label_dir)
        for current_root, _dirs, files in os.walk(label_dir):
            for name in files:
                path = os.path.join(current_root, name)
                ext = os.path.splitext(name)[1].lower()
                if ext not in SUPPORTED_EXT:
                    continue
                yield path, ext, label


def load_dataset(dataset_dir):
    rows = []
    labels = []
    for path, ext, label in iter_samples(dataset_dir):
        with open(path, "rb") as handle:
            data = handle.read()
        features = extract_features(data, ext, path, {})
        rows.append(feature_vector(features))
        labels.append(label)
    if not rows:
        raise SystemExit("no supported samples found under %s" % dataset_dir)
    if len(set(labels)) != 2:
        raise SystemExit("dataset must contain both benign and webshell samples")
    return rows, labels


def build_model():
    try:
        from deepforest import CascadeForestClassifier
        return CascadeForestClassifier(random_state=42), "deepforest.CascadeForestClassifier"
    except Exception:
        pass
    try:
        from sklearn.ensemble import RandomForestClassifier
        return RandomForestClassifier(n_estimators=200, random_state=42, class_weight="balanced"), "sklearn.RandomForestClassifier"
    except Exception:
        raise SystemExit(
            "install a training dependency first: deepforest or scikit-learn")


def main():
    args = parse_args()
    download_archives(args.download_url, args.allow_host, args.dataset_dir)
    rows, labels = load_dataset(args.dataset_dir)
    model, model_name = build_model()
    print("Training %s with %d samples and %d features" % (model_name, len(rows), len(FEATURE_NAMES)))
    model.fit(rows, labels)
    output_dir = os.path.dirname(os.path.abspath(args.output))
    os.makedirs(output_dir, exist_ok=True)
    with open(args.output, "wb") as handle:
        pickle.dump(model, handle, protocol=pickle.HIGHEST_PROTOCOL)
    print("Saved trusted local AST-DF model to %s" % args.output)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
